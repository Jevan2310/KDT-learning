## 모범 답안 (문제 4)
```javascript
const user = { name: "한코딩", goal: 8 };

const waterMessage = document.querySelector("#water-message");
const waterButton = document.querySelector("#water-button");

let waterCount = 0;

function drinkWater() {
  waterCount++;

  if (waterCount >= user.goal) {
    waterMessage.textContent = `${user.name}님, ${waterCount}잔 마셔서 목표 달성!`;
  } else {
    waterMessage.textContent = `물을 ${waterCount}잔 마셨어요.`;
  }
}

waterButton.addEventListener("click", drinkWater);
```
