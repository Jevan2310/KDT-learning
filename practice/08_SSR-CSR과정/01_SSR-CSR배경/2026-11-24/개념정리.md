# 개념 정리 — SSR-CSR과정 01. 서버에서 화면을 만들던 웹과 CSR이 확산된 배경

> 날짜: 2026-11-24 · wiki: [01_SSR-CSR배경](../../../../wiki/08_SSR-CSR과정/01_SSR-CSR배경.md)
> 단계별 설명문으로 쓰고, 단계 끝마다 직접 해 보는 예시를 둔다. 쉬운 말, 비유 없이. 예시는 이 Mac(Node.js)에서 미리 실행해 확인했다.

## 들어가며

학생 목록 화면은 이미 두 번 만들어 봤어요. Spring 03에서는 서버가 Thymeleaf로 학생 이름을 넣은 HTML을 만들어 보냈어요(`/students`). Spring 06에서는 같은 데이터를 JSON으로 보내고(`/api/students`), 브라우저의 React가 그걸 받아 목록을 그렸어요. 그때는 각각 만드는 데 집중해서, 두 방식이 **무엇이 다른지**는 따로 비교하지 않았어요.

SSR-CSR 과정은 이 둘을 같은 학생 목록·검색 기능으로 나란히 놓고 비교해요. 서버가 데이터를 넣은 HTML을 만들어 보내는 방식을 **SSR**, 브라우저의 JavaScript가 데이터로 화면을 만드는 방식을 **CSR**이라고 불러요. 오늘(01)은 코드를 만드는 회차가 아니에요. 두 방식의 이름과 차이, 그리고 CSR이 왜 많이 쓰이게 됐는지를 정리하는 회차예요.

<details><summary>server.js 전체 코드 (펼치기)</summary>

```javascript
// SSR·CSR 비교 예제 — Node.js에 들어 있는 http 모듈만 써요 (npm install 필요 없음)
// 실행: node server.js → 브라우저에서 http://localhost:3100/ssr, /csr, /api/students
const http = require("http");

const PORT = 3100;
const API_DELAY_MS = 0; // 6단계: 2000으로 바꾸면 /api/students 응답이 2초 늦게 가요

const students = [
  { id: 1, name: "한코딩", email: "han@example.com", className: "풀스택 1반" },
  { id: 2, name: "김자바", email: "kim@example.com", className: "풀스택 1반" },
  { id: 3, name: "이스프링", email: "lee@example.com", className: "풀스택 2반" },
  { id: 4, name: "박리액트", email: "park@example.com", className: "풀스택 2반" },
];

// 검색은 서버가 해요. 검색어가 ""이면 모든 이름에 들어 있다고 봐서 전체가 나와요
function findStudents(keyword) {
  return students.filter((student) => student.name.includes(keyword));
}

// 두 페이지가 같이 쓰는 문서 틀: 상단 메뉴, 펼치는 메뉴, 서버가 이 문서를 만든 시각
function page(title, main) {
  const madeAt = new Date().toTimeString().slice(0, 8);
  return `<!DOCTYPE html>
<html lang="ko">
<head><meta charset="UTF-8"><link rel="icon" href="data:,"><title>${title}</title></head>
<body>
<nav><a href="/ssr">SSR 목록</a> | <a href="/csr">CSR 목록</a> | <a href="/api/students">JSON</a></nav>
<details><summary>펼치는 메뉴</summary><p>검색한 뒤에도 열려 있을까요?</p></details>
<p>서버가 이 문서를 만든 시각: ${madeAt}</p>
<h1>${title}</h1>
${main}
</body>
</html>`;
}

const TABLE_HEAD = "<tr><th>번호</th><th>이름</th><th>이메일</th><th>반</th></tr>";

// SSR: 서버가 학생 이름을 넣은 <tr>까지 만들어서 보내요
function ssrPage(keyword) {
  const found = findStudents(keyword);
  let rows = found
    .map((s) => `<tr><td>${s.id}</td><td>${s.name}</td><td>${s.email}</td><td>${s.className}</td></tr>`)
    .join("\n");
  if (found.length === 0) rows = `<tr><td colspan="4">검색 결과가 없습니다.</td></tr>`;
  // 검색어는 사용자가 친 글자라서, HTML 기호로 읽히지 않게 바꿔서 넣어요
  const safeKeyword = keyword.replace(/&/g, "&amp;").replace(/"/g, "&quot;").replace(/</g, "&lt;");
  return page("학생 목록 (SSR)", `<form action="/ssr" method="get">
  <input name="keyword" value="${safeKeyword}" placeholder="학생 이름">
  <button type="submit">검색</button>
</form>
<table border="1">
<thead>${TABLE_HEAD}</thead>
<tbody>
${rows}
</tbody>
</table>`);
}

// CSR: 빈 표와 script만 보내요. 이름은 브라우저의 JavaScript가 /api/students로 받아서 넣어요
// (바깥 문자열이 백틱이라서 script 안에서는 백틱과 ${ }를 쓰지 않았어요)
function csrPage() {
  return page("학생 목록 (CSR)", `<form id="search-form">
  <input id="keyword" name="keyword" placeholder="학생 이름">
  <button type="submit">검색</button>
</form>
<p id="message"></p>
<table border="1">
<thead>${TABLE_HEAD}</thead>
<tbody id="rows"></tbody>
</table>
<script>
  const rows = document.querySelector("#rows");
  const message = document.querySelector("#message");

  async function loadStudents(keyword) {
    message.textContent = "불러오는 중...";
    rows.innerHTML = "";
    try {
      const response = await fetch("/api/students?keyword=" + encodeURIComponent(keyword));
      const students = await response.json();
      message.textContent = students.length === 0 ? "검색 결과가 없습니다." : "";
      for (const s of students) {
        const tr = document.createElement("tr");
        for (const value of [s.id, s.name, s.email, s.className]) {
          const td = document.createElement("td");
          td.textContent = value;
          tr.append(td);
        }
        rows.append(tr);
      }
    } catch (error) {
      message.textContent = "학생 목록을 불러오지 못했습니다.";
    }
  }

  document.querySelector("#search-form").addEventListener("submit", (event) => {
    event.preventDefault(); // 폼이 새 문서를 요청하지 못하게 막아요
    loadStudents(document.querySelector("#keyword").value.trim());
  });

  loadStudents(""); // 문서를 받자마자 전체 목록을 요청해요
</script>`);
}

const server = http.createServer((req, res) => {
  const url = new URL(req.url, "http://localhost:" + PORT);
  const keyword = (url.searchParams.get("keyword") ?? "").trim();
  if (url.pathname === "/favicon.ico") return res.end();
  // 어떤 요청이 왔는지 터미널에 한 줄씩 찍어요
  console.log(new Date().toTimeString().slice(0, 8), req.method, url.pathname, keyword ? "keyword=" + keyword : "");

  if (url.pathname === "/") {
    res.writeHead(302, { Location: "/ssr" });
    res.end();
  } else if (url.pathname === "/ssr") {
    res.writeHead(200, { "Content-Type": "text/html; charset=utf-8" });
    res.end(ssrPage(keyword));
  } else if (url.pathname === "/csr") {
    res.writeHead(200, { "Content-Type": "text/html; charset=utf-8" });
    res.end(csrPage());
  } else if (url.pathname === "/api/students") {
    setTimeout(() => {
      res.writeHead(200, { "Content-Type": "application/json; charset=utf-8" });
      res.end(JSON.stringify(findStudents(keyword)));
    }, API_DELAY_MS);
  } else {
    res.writeHead(404, { "Content-Type": "text/plain; charset=utf-8" });
    res.end("없는 주소예요: " + url.pathname);
  }
});

server.listen(PORT, () => {
  console.log("http://localhost:" + PORT + "/ssr 에서 시작해요. 끄려면 Ctrl + C");
});
```

## 1단계: 브라우저는 요청하고, 서버는 HTML을 응답한다

주소창에 주소를 치면 브라우저가 서버에 "이 주소의 내용을 주세요"라고 **요청**해요. 서버는 요청을 받아 필요한 일을 하고 **응답**을 보내요. 이때 서버가 보내는 건 화면을 찍은 이미지가 아니에요. HTML이라는 글자예요. 브라우저가 그 글자를 읽고 표와 글씨를 그려요. CSS·JavaScript·이미지 파일이 따로 있으면, 브라우저는 HTML을 읽은 뒤 그 파일들을 또 요청해요.

`/ssr`에서 서버는 학생 4명의 이름을 `<td>` 안에 넣은 HTML을 **완성해서** 보내요. 이렇게 서버가 데이터를 넣은 HTML을 만들어 응답하는 방식이 **SSR**(Server-Side Rendering)이에요. 렌더링은 데이터로 화면 구조를 만드는 일이에요. "서버에서 렌더링한다"고 해도 화면을 실제로 그리는 건 SSR에서도 브라우저예요. 서버는 이름이 들어간 HTML을 만드는 데까지만 해요.

Spring 03의 Thymeleaf가 바로 이 일을 했어요. `th:text="${student.name}"` 같은 템플릿 코드는 서버에서 처리되고, 브라우저는 `<td>한코딩</td>`처럼 값이 들어간 결과만 받아요. 브라우저는 Thymeleaf나 JSP 코드를 실행하지 않아요. `server.js`의 `ssrPage` 함수는 같은 일을 JavaScript 문자열로 해요. 학생 배열을 `map`으로 돌면서 `<tr>` 줄을 만들어 붙여요.

응답 맨 위에는 **헤더**가 붙어요. 본문이 아니라 응답에 대한 설명이에요. 그중 **Content-Type**은 본문이 어떤 종류인지 알려 줘요. `text/html`이면 브라우저가 HTML로 읽고 화면을 그려요. Content-Type이 틀리면 내용이 맞아도 화면이 이상하게 나와요.

`200 OK`는 요청이 성공했다는 뜻이에요. 빈 줄 위가 헤더, 아래가 본문이에요. 본문에 학생 이름 4개가 이미 들어 있어요. 서버 터미널에는 `15:26:45 GET /ssr` 줄이 찍혀요.

**바꿔 보기**
- `curl -s http://localhost:3100/ssr | grep -c 한코딩` → `1`. `-s`는 진행 표시를 숨기고, `grep -c`는 그 글자가 든 줄 수를 세요. 서버가 보낸 HTML 안에 이름이 있다는 뜻이에요.
- 브라우저의 `/ssr`에서 `Cmd + Option + U`(페이지 소스 보기)를 눌러요. curl과 같은 HTML이 보여요.
- `/ssr` 부분의 `"text/html; charset=utf-8"`을 `"text/plain; charset=utf-8"`로 바꾸고 서버를 다시 켜요. `curl -sI http://localhost:3100/ssr`에 `Content-Type: text/plain`이 나오고, 브라우저에서는 표 대신 `<tr><td>…` 태그가 글자 그대로 보여요. 받은 글자는 같은데 브라우저가 HTML로 읽지 않았기 때문이에요. 확인하고 되돌려요.

## 2단계: SSR에서 검색하면 새 문서를 받는다

서버가 페이지마다 HTML을 만들어 주는 웹에서는 링크를 누르거나 폼을 제출하면 **새 문서**를 요청하는 경우가 많아요. 문서는 브라우저가 화면에 띄우는 HTML 한 장이에요.

`/ssr`의 검색 폼은 `<form action="/ssr" method="get">`이고 입력칸에 `name="keyword"`가 있어요. `한`을 넣고 제출하면 브라우저가 `/ssr?keyword=한` 주소를 만들어 요청해요. `?` 뒤는 **쿼리 문자열**이라고 해요. 주소에 붙여 보내는 값이고, `name`이 값의 이름이 돼요. 서버는 `keyword`를 읽어 `findStudents`로 거르고, 결과를 넣은 HTML을 **처음부터 다시** 만들어 보내요. 브라우저는 지금 문서를 버리고 새 문서를 띄워요.

이 방식은 좋은 점이 분명해요. 서버에서 데이터와 HTML을 한 번에 만들기 쉬워요. 브라우저 쪽 JavaScript가 없어도 목록과 검색이 돼요. 검색 조건이 주소에 남아서, 주소를 공유하거나 나중에 다시 열면 같은 결과가 나와요. 단순한 목록·검색 화면이라면 이걸로 충분해요.

대신 바뀌는 게 목록뿐이어도 문서 전체가 새로 와요. 상단 메뉴도, 펼쳐 둔 메뉴도, 입력칸도 다시 만들어져요. 그래서 `ssrPage`는 검색어를 입력칸의 `value`에 다시 넣어 줘요. 이게 없으면 새 문서의 입력칸은 비어 있어요.

### ▶ 직접 해 보기

1. 브라우저 `/ssr`에서 **펼치는 메뉴**를 눌러 열고, "서버가 이 문서를 만든 시각"을 봐 둬요.
2. 검색칸에 `한`을 넣고 **검색**을 눌러요.
3. 주소창이 `localhost:3100/ssr?keyword=한`으로 바뀌어요. 시각이 바뀌고, 열어 둔 메뉴는 닫혀 있어요. 서버 터미널에는 문서 요청이 한 줄 찍혀요.

```text
15:26:52 GET /ssr keyword=한
```

curl로 같은 요청을 보내 봐요. curl 주소에는 한글을 그대로 넣으면 안 되고, `%ED%95%9C`처럼 주소용 글자로 바꿔서 넣어요. 브라우저는 이걸 알아서 해 줘요.

```bash
curl -s "http://localhost:3100/ssr?keyword=%ED%95%9C"
```

```text
…
<p>서버가 이 문서를 만든 시각: 15:26:52</p>
…
  <input name="keyword" value="한" placeholder="학생 이름">
…
<tbody>
<tr><td>1</td><td>한코딩</td><td>han@example.com</td><td>풀스택 1반</td></tr>
</tbody>
…
```

메뉴·시각·폼·표까지 문서 전체가 다시 왔어요. 입력칸의 `value="한"`도 서버가 넣었어요.

**바꿔 보기**
- 없는 이름을 검색해요. `curl -s "http://localhost:3100/ssr?keyword=%EC%B5%9C"`(`최`) → `<tr><td colspan="4">검색 결과가 없습니다.</td></tr>`. 안내 줄도 서버가 만들어 보내요.
- 한글을 바꾸기 귀찮으면 `curl -s -G --data-urlencode "keyword=한" http://localhost:3100/ssr`처럼 써요. curl이 바꿔서 붙여 줘요.
- 검색 결과 주소를 복사해서 새 탭에 붙여 넣어요. 같은 결과가 바로 나와요.

## 3단계: 필요한 데이터만 받아 일부만 바꾸기 — 부분 갱신

웹은 문서를 읽는 곳에서 일을 하는 곳으로 바뀌었어요. 검색어를 치는 동안 추천어가 뜨고, 메시지가 오면 대화 목록이 바뀌고, 장바구니 개수를 바꾸고, 글을 쓰는 중에 자동 저장되고, 지도를 옮기면 주변 정보가 바뀌어요. 이런 동작마다 새 문서를 받으면 화면이 계속 새로 고쳐져서 사용 흐름이 끊겨요. 2단계에서 본 것처럼 열어 둔 메뉴나 입력 중인 값도 사라져요.

학생 관리 화면에서 검색할 때 바꾸고 싶은 건 학생 목록뿐이에요. 그래서 나온 방법이 **JavaScript로 데이터만 요청해서, 바뀔 부분만 고치는 것**이에요. `fetch`는 비동기 과정과 Node 04에서 쓴 함수로, 페이지를 떠나지 않고 서버에 요청을 보내요. 이때 서버는 HTML 대신 **JSON**(데이터를 `{"이름": 값}` 모양의 글자로 적는 형식)을 보내요. 이렇게 화면이 아니라 데이터를 주고받으려고 만든 주소를 **API**라고 불러요. 오늘 예제에서는 `/api/students`예요.

문서를 그대로 두고 필요한 부분의 내용만 바꾸는 걸 **부분 갱신**이라고 해요. 문서 전체를 새로 요청하지 않을 뿐, 데이터가 필요하면 **서버에는 여전히 요청해요.** 그리고 부분 갱신에 React가 꼭 필요하지는 않아요. 일반 JavaScript로도 되고, SSR로 받은 페이지에도 붙일 수 있어요.

### ▶ 직접 해 보기

```bash
curl -i http://localhost:3100/api/students
```

```text
HTTP/1.1 200 OK
Content-Type: application/json; charset=utf-8
…

[{"id":1,"name":"한코딩","email":"han@example.com","className":"풀스택 1반"},{"id":2,"name":"김자바","email":"kim@example.com","className":"풀스택 1반"},{"id":3,"name":"이스프링","email":"lee@example.com","className":"풀스택 2반"},{"id":4,"name":"박리액트","email":"park@example.com","className":"풀스택 2반"}]
```

Content-Type이 `application/json`이고, 본문에 태그가 하나도 없어요. 데이터만 있어요. 검색도 돼요.

```bash
curl -s "http://localhost:3100/api/students?keyword=%EA%B9%80"
```

```text
[{"id":2,"name":"김자바","email":"kim@example.com","className":"풀스택 1반"}]
```

이제 SSR 페이지에서 부분 갱신을 직접 해 봐요. 브라우저 `/ssr`(검색어 없이)을 열고 **펼치는 메뉴**를 연 다음, `Cmd + Option + J`로 Console을 열고 아래를 붙여 넣어 Enter를 눌러요.

```javascript
const response = await fetch("/api/students?keyword=김");
const found = await response.json();
document.querySelector("tbody").innerHTML = found
  .map((s) => `<tr><td>${s.id}</td><td>${s.name}</td><td>${s.email}</td><td>${s.className}</td></tr>`)
  .join("");
```

표에 김자바만 남아요. 그런데 주소는 `/ssr` 그대로이고, 시각도 그대로이고, 메뉴도 열려 있어요. 서버 터미널에는 문서 요청이 아니라 데이터 요청만 찍혀요.

```text
15:27:28 GET /api/students keyword=김
```

**바꿔 보기**
- 이 상태에서 `Cmd + Option + U`(페이지 소스 보기)를 열어요. 이름이 여전히 4개예요. 페이지 소스는 서버가 처음 보낸 HTML이라서, JavaScript가 나중에 바꾼 건 안 보여요.
- 다시 해 보려면 새로고침하고, `김`을 `이`로 바꿔 붙여 넣어요. 이스프링만 남아요.

## 4단계: CSR — 처음 받은 HTML에는 이름이 없다

**CSR**(Client-Side Rendering)은 브라우저(클라이언트)의 JavaScript가 데이터로 화면 구조를 만드는 방식이에요. 클라이언트는 요청을 보내는 쪽, 여기서는 브라우저예요.

`/csr`에서 서버가 보내는 HTML에는 표의 머리와 **빈 `<tbody id="rows"></tbody>`**, 그리고 `<script>`만 있어요. 학생 이름은 없어요. 브라우저는 이 HTML을 그린 다음 script를 실행해요. script의 `loadStudents("")`가 `/api/students`에 요청하고, 받은 JSON으로 `<tr>`을 만들어 `tbody`에 넣어요. 그때서야 이름이 보여요. 순서로 쓰면 이래요.

```text
① HTML(빈 표 + script) 받기 → ② 브라우저에서 script 실행 → ③ 학생 목록 API 요청
→ ④ JSON 받기 → ⑤ script가 학생 목록 <tr> 만들기
```

React도 같아요. Spring 06의 React 화면도 처음 HTML은 `<div id="root"></div>`와 `<script type="module" src="/src/main.jsx">` 정도였고, React가 실행된 뒤 API 응답으로 목록을 만들었어요. 오늘 예제는 React 대신 `createElement`로 직접 만들었을 뿐이에요.

같은 한코딩을 보여 줘도 서버의 응답이 달라요. SSR은 `<td>한코딩</td>`이 든 **HTML**을, CSR의 목록 API는 `{"id":1,"name":"한코딩",…}` 같은 **JSON**을 보내요. 학생 목록 HTML은 SSR에서는 서버가, CSR에서는 브라우저의 JavaScript가 만들어요. 화면 모양은 SSR에서는 서버 템플릿과 CSS가, CSR에서는 JavaScript(React) 코드와 CSS가 정해요. 하지만 학생 데이터를 찾는 건 **둘 다 서버**예요(`findStudents`, Spring에서는 Service와 DB). 같은 JSON은 브라우저뿐 아니라 모바일 앱도 받아 쓸 수 있어요.

### ▶ 직접 해 보기

```bash
curl -s http://localhost:3100/csr | grep -c 한코딩
curl -s http://localhost:3100/ssr | grep -c 한코딩
curl -s http://localhost:3100/csr | grep tbody
```

```text
0
1
<tbody id="rows"></tbody>
```

`/csr`이 보낸 HTML에는 이름이 0번 나와요. 이제 브라우저에서 `http://localhost:3100/csr`을 열어요. 표에 4명이 보이고, 서버 터미널에는 요청이 **두 번** 찍혀요. 문서 요청, 그리고 script가 보낸 데이터 요청이에요.

```text
15:27:24 GET /csr
15:27:25 GET /api/students
```

**바꿔 보기**
- `Cmd + Option + U`(페이지 소스 보기)를 열어요. `<tbody id="rows"></tbody>`가 비어 있어요. 이번에는 `Cmd + Option + I` → **Elements** 탭에서 `tbody`를 펼쳐요. `<tr>` 4개가 들어 있어요. 페이지 소스는 서버가 보낸 HTML, Elements는 JavaScript가 실행된 뒤 지금 화면의 구조예요 (SEO 03에서 본 차이).
- 개발자 도구를 연 채로 `Cmd + Shift + P` → `Disable JavaScript` → 새로고침해요. `/csr`은 표 머리만 있고 학생이 없어요. `/ssr`은 목록도 검색도 그대로 돼요. 다 봤으면 같은 방법으로 `Enable JavaScript`를 해요.

## 5단계: CSR 검색 — 문서는 두고 목록만 바꾼다

`/csr`의 검색 폼에는 `action`이 없어요. 대신 script가 폼의 `submit` 이벤트를 받아서 처리해요.

```javascript
document.querySelector("#search-form").addEventListener("submit", (event) => {
  event.preventDefault(); // 폼이 새 문서를 요청하지 못하게 막아요
  loadStudents(document.querySelector("#keyword").value.trim());
});
```

폼은 원래 제출되면 새 문서를 요청해요(2단계). `event.preventDefault()`가 이 기본 동작을 막아요(React 08 할 일 앱의 폼에서 쓴 것과 같아요). 그다음 `loadStudents`가 `fetch`로 JSON만 받아 표 내용만 바꿔요. 그래서 주소도, 시각도, 열어 둔 메뉴도, 입력칸 값도 그대로예요.

여기서 "CSR이면 서버가 할 일이 줄어든다"고 생각하기 쉬운데, 그렇지 않아요. 검색어로 학생을 거르는 `findStudents`는 여전히 **서버**에서 돌아요. Spring에서도 학생 데이터 조회, 검색 조건 처리, 입력값 검사, 접근 권한 확인, 데이터 변경은 계속 서버가 맡아요. 바뀌는 건 **목록 HTML을 만드는 곳**(서버 → 브라우저)과 **서버가 보내는 응답의 형태**(HTML → JSON)예요.

CSR의 좋은 점은 네 가지예요. 첫째, 방금 본 것처럼 문서를 두고 필요한 부분만 바꿀 수 있어요. 둘째, 상호작용이 많은 화면을 만들기 좋아요. 오늘 예제는 "불러오는 중"·"결과 없음"을 손으로 바꿨지만, React는 검색어·선택한 학생·로딩 여부·오류 메시지를 **state**로 두고 화면을 state에 맞춰 일관되게 바꿔요. 셋째, 화면(React)과 데이터 API(Spring)의 역할을 나눌 수 있어요. 다만 API를 여러 곳에서 쓰는 건 CSR만의 기능이 아니에요. 3단계에서 SSR 페이지도 같은 API를 썼어요. 넷째, 앱 코드를 한 번 받은 뒤에는 작은 JSON만 받아서 이후 동작이 빠르게 느껴지게 할 수 있어요. 하지만 CSR이라고 모든 동작이 빨라지지는 않아요. API 응답 시간, JavaScript 실행량, 네트워크 상태에 따라 달라요.

### ▶ 직접 해 보기

1. 브라우저 `/csr`에서 **펼치는 메뉴**를 열고 시각을 봐 둬요.
2. 검색칸에 `이`를 넣고 **검색**을 눌러요. 이스프링만 남아요.
3. 주소는 `/csr` 그대로, 시각도 그대로, 메뉴는 열려 있고, 검색칸에 `이`가 남아 있어요. 서버 터미널에는 데이터 요청 한 줄만 찍혀요.

```text
15:27:26 GET /api/students keyword=이
```

4. `최`를 검색하면 표가 비고 위에 `검색 결과가 없습니다.`가 나와요.

**바꿔 보기**
- `event.preventDefault();` 줄 앞에 `//`를 붙여 주석으로 만들고, 서버를 다시 켜고 새로고침한 뒤 `한`을 검색해요. 주소가 `/csr?keyword=한`으로 바뀌고 시각도 바뀌고, 학생 4명이 모두 보여요. 서버 터미널에는 `GET /csr keyword=한`이 찍혀요. 폼이 원래대로 새 문서를 요청했는데, `/csr`은 검색어와 상관없이 늘 같은 빈 틀을 보내고 script가 전체 목록부터 다시 불러왔기 때문이에요. 확인하고 되돌려요.

## 6단계: CSR에서 더 챙겨야 할 것

CSR에는 비용도 있어요. 첫 화면은 HTML을 받고, JavaScript를 실행하고, API 응답까지 기다려야 목록이 나와요. 이게 길어지면 **빈 영역이 오래 보여요.** 그래서 네 가지 상태를 일부러 나눠서 보여 줘야 해요. 요청 중이면 로딩 안내, 성공하면 목록, 결과가 없으면 빈 결과 안내, 실패하면 오류 안내예요. 오늘 `loadStudents`는 `message` 한 줄로 이 넷을 처리해요. Node 04와 비동기 04에서 `isLoading`·`error`·빈 배열 검사로 만든 것과 같은 일이에요.

JavaScript를 받고 해석하고 실행하는 데는 브라우저의 시간이 들어요. 기기 성능이 낮으면 더 느려져요. 검색엔진도 생각해야 해요. 4단계의 curl처럼 처음 HTML만 읽는 프로그램은 `/csr`에서 이름을 하나도 못 봐요. SEO 03에서 네이버 소유확인이 JavaScript를 실행하지 않아서 태그를 처음 HTML에 넣었던 것도 같은 이유예요. 그렇다고 "CSR 페이지는 검색에 절대 안 나온다"고 단정하면 안 돼요. 검색엔진마다 JavaScript를 처리하는 방식과 시점이 달라서, 공개해야 할 내용을 어떤 형태로 보낼지 고민해야 한다는 뜻이에요.

마지막으로 직접 챙길 상태가 늘어나요. 데이터 요청과 화면 표시가 나뉘어서, 요청 실패, 로딩 중 중복 요청, 화면과 서버 데이터가 서로 다른 상황, 페이지를 옮길 때 상태를 유지하는 일을 함께 생각해야 해요. 그래서 SSR과 CSR은 어느 쪽이 항상 낫다기보다, 장점과 비용이 다른 선택지예요.

### ▶ 직접 해 보기

`server.js`의 `const API_DELAY_MS = 0;`을 `2000`으로 바꾸고 서버를 다시 켜요. 브라우저 `/csr`을 새로고침해요. 2초 동안 `불러오는 중...`과 빈 표만 보이다가 학생 4명이 나와요. `/ssr`은 API를 쓰지 않아서 바로 나와요.

**바꿔 보기**
- 2초 지연인 채로 `한`을 넣고 **검색**을 빠르게 두 번 눌러요. 한코딩이 **두 줄** 나와요. 두 요청이 각각 시작할 때 표를 비우고, 응답이 올 때마다 줄을 붙였기 때문이에요. 강의가 말한 "로딩 중 중복 요청"이에요. 실제 화면에서는 로딩 중에 버튼을 막는 식으로 처리해요. 확인하면 `0`으로 되돌려요.
- `/csr`을 열어 둔 채 서버 터미널에서 `Ctrl + C`로 서버를 꺼요. 검색하면 `학생 목록을 불러오지 못했습니다.`가 나와요. `fetch`가 실패해서 `catch`로 갔어요. Console에는 `net::ERR_CONNECTION_REFUSED`가 든 빨간 줄이 보여요. 같은 상태에서 `/ssr` 검색은 Chrome의 "사이트에 연결할 수 없음" 화면이 나와요. 다 보면 서버를 다시 켜요.

## 7단계: 부분 갱신 · SPA · CSR은 기준이 다르다

세 말은 비슷한 곳에서 나오지만 뜻이 같지 않아요. 묻는 질문이 달라요.
- **부분 갱신**: 화면의 **일부**만 바꾸는가?
- **SPA**(Single-Page Application): 기본 문서를 **한 번** 받은 뒤, 그 문서를 유지하면서 JavaScript로 주요 화면을 바꾸는가?
- **CSR**: 브라우저의 **JavaScript가 화면 구조**를 만드는가?

오늘 예제에 대 보면 이래요. 3단계는 SSR로 받은 페이지에 부분 갱신을 붙인 거예요. `/csr`은 CSR이고 검색은 부분 갱신이지만 SPA는 아니에요. 화면이 하나뿐이고, 위 메뉴의 링크를 누르면 새 문서를 받기 때문이에요. SPA는 학생 목록·상세·등록처럼 **여러 화면**을 한 문서 안에서 바꿔 보여 주는 구조예요. 이름에 'Single Page'가 있지만 화면이 하나라는 뜻이 아니고, 주소도 바뀔 수 있어요. 팀 프로젝트의 채팅 앱이 React Router로 로그인·방 목록·채팅방 화면을 바꾸는 게 SPA예요. SPA는 CSR로 많이 만들지만, 첫 HTML을 SSR로 만들 수도 있어요. React도 CSR만 하는 도구가 아니에요(05회차).

기술 이름도 같은 칸에 놓이지 않아요. Spring Boot는 요청을 받는 서버 애플리케이션이에요. JSP와 Thymeleaf는 서버에서 HTML을 만드는 템플릿이에요. React는 브라우저에서 화면과 상태를 다뤄요. 그래서 JSP를 Thymeleaf로 바꾸면 템플릿 기술만 바뀌고 **SSR이라는 방식은 그대로**예요. 오늘 `ssrPage`처럼 JavaScript 문자열로 만들어도 SSR이에요. SSR이냐 CSR이냐는 쓰는 도구가 아니라 **데이터가 든 HTML을 어디서 만드느냐**로 정해져요.

```text
JSP SSR:        Spring → JSP → HTML → 브라우저
Thymeleaf SSR:  Spring → Thymeleaf → HTML → 브라우저
React CSR:      Spring → JSON → 브라우저의 React → 화면 구성
오늘 예제:       /ssr → 이름 든 HTML · /csr → 빈 표 + script · /api/students → JSON
```

### ▶ 직접 해 보기

브라우저 `/csr`에서 위 메뉴의 **SSR 목록**을 눌러요. 서버 터미널에 `GET /ssr`이 찍히고 시각이 바뀌어요. 링크로 새 문서를 받았어요. 다시 **CSR 목록**을 누르면 `GET /csr`과 `GET /api/students` 두 줄이 찍혀요. 문서가 바뀔 때마다 처음부터 다시 받는 거라 SPA가 아니에요.

**바꿔 보기**
- 이 셋을 직접 나눠 봐요. ① 2단계 `/ssr` 검색 ② 3단계 Console로 바꾼 `/ssr` 표 ③ 5단계 `/csr` 검색. 문서를 새로 받았나(①만), 일부만 바꿨나(②③), 첫 목록 HTML을 브라우저 JavaScript가 만들었나(③만)를 따져 보면 돼요.

## 마무리: 오늘 예제에서 일어나는 일, 순서대로

1. `node server.js` → 서버가 3100번에서 요청을 기다린다.
2. 브라우저가 `/ssr`을 요청한다 → 서버가 `findStudents("")`로 4명을 찾고, 이름을 넣은 `<tr>`까지 HTML을 만들어 `text/html`로 응답한다 → 브라우저가 그 HTML을 읽어 표를 그린다.
3. `/ssr`에서 `한`을 검색한다 → 폼이 `/ssr?keyword=한` 새 문서를 요청한다 → 서버가 HTML 전체를 다시 만든다 → 브라우저가 문서를 통째로 바꾼다 (주소에 검색어, 시각 바뀜, 메뉴 닫힘).
4. 브라우저가 `/csr`을 요청한다 → 서버가 이름 없는 빈 표와 script를 응답한다.
5. 브라우저가 HTML을 그리고 script를 실행한다 → `불러오는 중...` → `fetch("/api/students")`.
6. 서버가 `findStudents("")` 결과를 `application/json`으로 응답한다.
7. script가 JSON으로 `<tr>`을 만들어 `tbody`에 넣는다 → 이름이 보인다. 페이지 소스에는 여전히 없다.
8. `/csr`에서 `이`를 검색한다 → `preventDefault`가 새 문서 요청을 막는다 → `fetch("/api/students?keyword=이")` → 서버가 거른 JSON → script가 표만 바꾼다 (주소·시각·메뉴 그대로).
9. 응답이 늦거나 서버가 꺼져 있으면 → script가 로딩·오류 문구로 지금 상태를 알려 준다.

## 다음 회차와 연결

02회차는 오늘 `/ssr`을 Spring Boot와 JSP로 만들어요. `@GetMapping("/students")`가 요청을 받고, `@RequestParam`으로 `keyword`를 읽고, `Model`에 학생 목록을 담아요. JSP의 `<c:forEach>`가 오늘 `ssrPage`의 `map`처럼 서버에서 `<tr>`을 반복해서 만들어요. 03회차는 오늘 서버 터미널로 본 요청을 Chrome 개발자 도구 Network에서 봐요. 문서 요청(Doc)의 Response에 이름이 들어 있는지, 검색하면 새 문서 요청이 생기는지, JavaScript를 꺼도 되는지 확인해요. 04회차는 오늘 `/csr`과 `/api/students`를 React와 `@RestController`로 만들어요. 오늘 `loadStudents`가 손으로 하던 로딩·오류·빈 결과 처리를 `useState`와 `useEffect`로 해요.

## 오류가 날 때 찾아보기

| 오류 문장 · 증상 | 먼저 볼 곳 |
|---|---|
| `Error: listen EADDRINUSE: address already in use :::3100` | 3100번에서 서버가 이미 켜져 있음 → 켜 둔 터미널에서 `Ctrl + C` (시작하기) |
| `curl: (7) Failed to connect to localhost port 3100 …` / 브라우저 "사이트에 연결할 수 없음" | 서버가 꺼져 있음 → `node server.js` (시작하기, 6단계) |
| curl 주소에 한글을 그대로 넣었더니 `HTTP/1.1 400 Bad Request`, 본문 없음 | 한글을 `%ED%95%9C`처럼 바꾸거나 `curl -G --data-urlencode "keyword=한" …` (2단계) |
| `없는 주소예요: /students` | 오늘 주소는 `/ssr`·`/csr`·`/api/students`. `/students`는 Spring 프로젝트 주소 |
| 코드를 고쳤는데 그대로 | 서버를 `Ctrl + C`로 끄고 다시 `node server.js` (시작하기) |
| 표 대신 `<tr><td>…`가 글자로 보임 | Content-Type이 `text/plain` → `text/html`로 (1단계) |
| 서버가 안 켜지고 `SyntaxError: Unexpected token '{'` | `csrPage`의 script 안에 백틱을 씀 → 바깥 백틱 문자열이 거기서 끝남. `"…" + 값`으로 쓰기 (4단계) |
| `/csr`을 열면 서버가 `ReferenceError: keyword is not defined`로 꺼지고 curl은 `(52) Empty reply from server` | script 안에 `${ }`를 씀 → Node가 서버에서 먼저 계산하려 함. `"…" + 값`으로 쓰기 (4단계) |
| CSR 검색 뒤 주소가 `/csr?keyword=…`로 바뀌고 전체 목록이 나옴 | `event.preventDefault()`가 빠짐 (5단계) |
| `/csr`에 `학생 목록을 불러오지 못했습니다.` | 서버가 꺼짐 또는 API 오류 → 서버 터미널, Console 빨간 줄 (6단계) |
| `불러오는 중...`이 오래 보임 | `API_DELAY_MS` 값 (6단계) |
| 같은 학생이 두 줄 나옴 | 로딩 중에 검색을 두 번 누름 (6단계) |
| 페이지 소스에는 이름이 없는데 화면에는 있음 | CSR이라 정상. JavaScript가 나중에 넣음 → Elements에서 보기 (4단계) |
| Console에 붙여 넣기가 막힘 | Chrome 안내 문구에 나온 허용 문장을 그대로 쳐서 허용 (3단계) |
